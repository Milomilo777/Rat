private let logger = Logger(subsystem: "com.imanmontajabi.Rat", category: "alerts")

struct ConnectionAlert: Identifiable, Equatable {
    let id = UUID()
    let appID: String
    let appName: String
    let host: String
    let address: String
    let port: Int
    let proto: String
    var key: String { "\(appID)|\(registrableDomain(host))" }
}

/// Queues unknown connections and shows them one at a time in a floating panel.
@MainActor
final class AlertCenter: ObservableObject {
    @Published private(set) var current: ConnectionAlert?
    private var queue: [ConnectionAlert] = []
    private var answeredOrQueued = Set<String>()
    private var panel: NSPanel?
    private unowned let rules: RulesModel
    private let onDecision: (ConnectionAlert, Verdict) -> Void
    var iconProvider: (String) -> NSImage = { _ in NSImage() }
    /// Set by the ledger: false once rules, a policy button or the mode have already answered this alert.
    var isStillPending: (ConnectionAlert) -> Bool = { _ in true }
    private var timeoutTask: Task<Void, Never>?

    init(rules: RulesModel, onDecision: @escaping (ConnectionAlert, Verdict) -> Void) {
        self.rules = rules
        self.onDecision = onDecision
    }

    var pendingCount: Int { queue.count + (current == nil ? 0 : 1) }

    func report(_ alert: ConnectionAlert) {
        guard answeredOrQueued.insert(alert.key).inserted else { return }
        queue.append(alert)
        logger.notice("alert queued for \(alert.appName, privacy: .public)")
        presentNext()
    }

    /// Drops queued alerts that no longer need an answer (e.g. the app got a process-wide Allow or Deny).
    func revalidate() {
        let before = queue.count
        queue.removeAll { a in
            let keep = isStillPending(a)
            if !keep { answeredOrQueued.remove(a.key) }
            return !keep
        }
        if let c = current, !isStillPending(c) {
            timeoutTask?.cancel()
            answeredOrQueued.remove(c.key)
            current = nil
            closePanel()
        }
        if before != queue.count || current == nil { presentNext() }
    }

    func answer(_ alert: ConnectionAlert, verdict: Verdict, scope: RulesModel.Scope, lifetime: Lifetime) {
        timeoutTask?.cancel()
        rules.addRule(from: alert, verdict: verdict, scope: scope, lifetime: lifetime)
        onDecision(alert, verdict)
        // A "once" answer should not silence the next connection from the same app.
        if lifetime == .once { answeredOrQueued.remove(alert.key) }
        current = nil
        closePanel()
        presentNext()
    }

    private func presentNext() {
        guard current == nil, !queue.isEmpty else { return }
        // Skip anything answered while it waited in the queue.
        while let next = queue.first, !isStillPending(next) { answeredOrQueued.remove(next.key); queue.removeFirst() }
        guard !queue.isEmpty else { return }
        current = queue.removeFirst()
        showPanel()
        startTimeout()
    }

    /// Settings > Alert: confirm automatically after N seconds.
    private func startTimeout() {
        let secs = AppSettings.shared.alertTimeout
        guard secs > 0, let alert = current else { return }
        timeoutTask = Task { [weak self] in
            try? await Task.sleep(for: .seconds(secs))
            guard !Task.isCancelled, let self, self.current == alert else { return }
            let allow = AppSettings.shared.alertTimeoutAllows
            if AppSettings.shared.notifyAlertTimeout {
                Notifier.post(title: allow ? "Connection allowed after timeout" : "Connection denied after timeout", body: "\(alert.appName) → \(alert.host)")
            }
            self.answer(alert, verdict: allow ? .allow : .deny, scope: .any, lifetime: .once)
        }
    }

    private func showPanel() {
        guard let alert = current else { return }
        let view = AlertView(alert: alert, icon: iconProvider(alert.appID), remaining: queue.count) { [weak self] verdict, scope, lifetime in
            self?.answer(alert, verdict: verdict, scope: scope, lifetime: lifetime)
        }
        let host = NSHostingView(rootView: view.preferredColorScheme(.dark))
        host.setFrameSize(host.fittingSize)
        let p = NSPanel(contentRect: NSRect(origin: .zero, size: host.fittingSize),
                        styleMask: [.titled, .fullSizeContentView, .nonactivatingPanel, .hudWindow, .utilityWindow],
                        backing: .buffered, defer: false)
        p.titleVisibility = .hidden
        p.titlebarAppearsTransparent = true
        p.isFloatingPanel = true
        p.level = .floating
        p.hidesOnDeactivate = false
        p.isMovableByWindowBackground = true
        p.collectionBehavior = [.canJoinAllSpaces, .fullScreenAuxiliary]
        p.contentView = host
        if let screen = NSScreen.main {
            p.setFrameOrigin(NSPoint(x: screen.visibleFrame.maxX - p.frame.width - 20, y: screen.visibleFrame.maxY - p.frame.height - 20))
        }
        p.orderFrontRegardless()
        panel = p
    }

    private func closePanel() { panel?.orderOut(nil); panel = nil }
}

struct AlertView: View {
    let alert: ConnectionAlert
    let icon: NSImage
    let remaining: Int
    let decide: (Verdict, RulesModel.Scope, Lifetime) -> Void

    @ObservedObject private var settings = AppSettings.shared
    // One click should cover the app, not just one of its many server addresses.
    @State private var scope: RulesModel.Scope = .any
    @State private var lifetime: Lifetime = .forever

    private var target: String {
        alert.host == alert.address ? alert.address : alert.host
    }

    var body: some View {
        VStack(alignment: .leading, spacing: 12) {
            HStack(spacing: 12) {
                Image(nsImage: icon).resizable().frame(width: 44, height: 44)
                VStack(alignment: .leading, spacing: 2) {
                    Text(alert.appName).font(.title3.weight(.semibold))
                    Text("wants to connect to").font(.callout).foregroundStyle(.secondary)
                }
                Spacer()
                if remaining > 0 { Text("+\(remaining)").font(.caption).padding(6).background(.white.opacity(0.12), in: Capsule()) }
            }
            VStack(alignment: .leading, spacing: 2) {
                Text(target).font(.headline).textSelection(.enabled)
                Text("\(alert.proto) port \(alert.port) · \(alert.address)").font(.caption).foregroundStyle(.secondary)
            }
            Divider()
            Picker("Applies to", selection: $scope) { ForEach(RulesModel.Scope.allCases) { Text($0.rawValue).tag($0) } }
            Picker("Keep rule", selection: $lifetime) {
                Text("Once").tag(Lifetime.once); Text("Until Quit").tag(Lifetime.untilQuit); Text("Forever").tag(Lifetime.forever)
            }.pickerStyle(.segmented)
            Text("The connection has already started. Deny cuts it off and blocks the server address.")
                .font(.caption).foregroundStyle(.secondary).fixedSize(horizontal: false, vertical: true)
            HStack {
                Button { decide(.deny, scope, lifetime) } label: { Label("Deny", systemImage: "xmark").frame(maxWidth: .infinity) }
                    .modifier(OptionalShortcut(key: .cancelAction, enabled: settings.confirmWithReturn)).tint(.red)
                Button { decide(.allow, scope, lifetime) } label: { Label("Allow", systemImage: "checkmark").frame(maxWidth: .infinity) }
                    .modifier(OptionalShortcut(key: .defaultAction, enabled: settings.confirmWithReturn)).buttonStyle(.borderedProminent).tint(.green)
            }
            .controlSize(.large)
        }
        .padding(18)
        .frame(width: 380)
        .background(.regularMaterial)
    }
}

/// "Confirm with Return and Escape" can be turned off to avoid accidental answers.
struct OptionalShortcut: ViewModifier {
    let key: KeyboardShortcut
    let enabled: Bool
    @ViewBuilder func body(content: Content) -> some View {
        if enabled { content.keyboardShortcut(key) } else { content }
    }
}

    @Published var notifyProfileSwitch: Bool { didSet { defaults.set(notifyProfileSwitch, forKey: "notifyProfileSwitch") } }
    @Published var notifyAlertTimeout: Bool { didSet { defaults.set(notifyAlertTimeout, forKey: "notifyAlertTimeout") } }

    @Published var autoCheckUpdates: Bool { didSet { defaults.set(autoCheckUpdates, forKey: "autoCheckUpdates") } }
    @Published var showPreReleases: Bool { didSet { defaults.set(showPreReleases, forKey: "showPreReleases") } }
    @Published var lastUpdateCheck: Date? { didSet { defaults.set(lastUpdateCheck, forKey: "lastUpdateCheck") } }
    @Published var lastUpdateCheckFailed: Bool { didSet { defaults.set(lastUpdateCheckFailed, forKey: "lastUpdateCheckFailed") } }

    // Advanced
    @Published var approveRulesAutomatically: Bool { didSet { defaults.set(approveRulesAutomatically, forKey: "approveRulesAutomatically") } }

    private init() {
        func bool(_ k: String, _ d: Bool) -> Bool { UserDefaults.standard.object(forKey: k) as? Bool ?? d }
        showInMenuBar = bool("showInMenuBar", true)
        numericRates = bool("numericRates", true)
        silentModeSymbol = bool("silentModeSymbol", true)
        trafficRateUnit = RateUnit(rawValue: UserDefaults.standard.string(forKey: "trafficRateUnit") ?? "") ?? .bytes
        confirmWithReturn = bool("confirmWithReturn", true)
        alertTimeout = UserDefaults.standard.integer(forKey: "alertTimeout")
        alertTimeoutAllows = bool("alertTimeoutAllows", false)
        includeLocalNetwork = bool("includeLocalNetwork", false)
        includeNetworkExtensions = bool("includeNetworkExtensions", false)
        includeOtherUsers = bool("includeOtherUsers", false)
        groupBy = GroupBy(rawValue: UserDefaults.standard.string(forKey: "groupBy") ?? "") ?? .process
        sortKey = SortKey(rawValue: UserDefaults.standard.string(forKey: "sortKey") ?? "") ?? .recent
        floatOnTop = bool("floatOnTop", false)
        additionalLocalRanges = UserDefaults.standard.string(forKey: "additionalLocalRanges") ?? ""
        notifySilentConnections = bool("notifySilentConnections", true)
        notifyProfileSwitch = bool("notifyProfileSwitch", true)
        notifyAlertTimeout = bool("notifyAlertTimeout", true)
        autoCheckUpdates = bool("autoCheckUpdates", true)
        showPreReleases = bool("showPreReleases", false)
        lastUpdateCheck = UserDefaults.standard.object(forKey: "lastUpdateCheck") as? Date
        lastUpdateCheckFailed = bool("lastUpdateCheckFailed", false)
        approveRulesAutomatically = bool("approveRulesAutomatically", true)
        LocalRanges.shared.set(additionalLocalRanges)
    }

    /// Rate formatted in the chosen unit, e.g. "1.2 MB/s" or "9.6 Mbit/s".
    func formatRate(_ bytesPerSecond: Double) -> String {
        switch trafficRateUnit {
        case .bytes:
            let f = ByteCountFormatter(); f.countStyle = .file; f.allowsNonnumericFormatting = false
            return f.string(fromByteCount: Int64(bytesPerSecond)) + "/s"
        case .bits:
            let bits = bytesPerSecond * 8
            for (unit, div) in [("Gbit/s", 1e9), ("Mbit/s", 1e6), ("kbit/s", 1e3)] where bits >= div { return String(format: "%.1f %@", bits / div, unit) }
            return "\(Int(bits)) bit/s"
        }
    }
}

/// Extra address ranges the user wants treated as the local network ("Additional IP Ranges in Local Network").
final class LocalRanges: @unchecked Sendable {
    static let shared = LocalRanges()
    private let lock = NSLock()
    private var ranges: [(base: UInt32, mask: UInt32)] = []

    /// Parses "10.5.0.0/16, 100.64.0.0/10" (comma, space or newline separated). Returns the entries that did not parse.
    @discardableResult
    func set(_ text: String) -> [String] {
        var bad: [String] = []
        var parsed: [(UInt32, UInt32)] = []
        for token in text.split(whereSeparator: { ", \n".contains($0) }) {
            if let r = Self.parse(String(token)) { parsed.append(r) } else { bad.append(String(token)) }
        }
        lock.withLock { ranges = parsed }
        return bad
    }

    func contains(_ address: String) -> Bool {
        var a = in_addr()
        guard inet_pton(AF_INET, address, &a) == 1 else { return false }
        let ip = UInt32(bigEndian: a.s_addr)
        return lock.withLock { ranges.contains { (ip & $0.mask) == $0.base } }
    }

    static func parse(_ s: String) -> (base: UInt32, mask: UInt32)? {
        let parts = s.split(separator: "/", omittingEmptySubsequences: false)
        guard parts.count <= 2 else { return nil }
        var a = in_addr()
        guard inet_pton(AF_INET, String(parts[0]), &a) == 1 else { return nil }
        let bits = parts.count == 2 ? Int(parts[1]) : 32
        guard let bits, (0...32).contains(bits) else { return nil }
        let mask: UInt32 = bits == 0 ? 0 : ~UInt32(0) << UInt32(32 - bits)
        return (UInt32(bigEndian: a.s_addr) & mask, mask)
    }
}

/// Applies "deny" decisions with the macOS packet filter.
///
/// Rat has no Network Extension entitlement, so it cannot intercept connections per process.
/// Instead it collects the server addresses of denied connections and loads them into a `pf` table
/// through the bundled `rat-pf-helper.sh`, which runs as root. Two ways to run it:
///
/// * **Manual** (default): each click on "Apply blocking" runs the helper once behind the standard
///   macOS password prompt. Nothing stays installed.
/// * **Automatic** (opt-in, explicitly approved by the user): the helper is installed once as two root
///   LaunchDaemons. One re-applies the blocklist whenever Rat rewrites it, so Deny takes effect
///   immediately; the other snapshots all processes' sockets so Rat can see system-wide traffic.
///   "Remove Helper" uninstalls both and clears the pf anchor.
///
/// A blocked address is blocked for every app on the Mac.
@MainActor
final class Enforcer: ObservableObject {
    @Published private(set) var applied: Set<String> = []
    @Published private(set) var busy = false
    @Published private(set) var helperInstalled: Bool
    @Published var autoApply: Bool { didSet { UserDefaults.standard.set(autoApply, forKey: "autoApplyPF") } }
    @Published var lastError: String?

    static let applyLabel = "com.imanmontajabi.rat.pf"
    static let samplerLabel = "com.imanmontajabi.rat.sampler"
    static let helperPath = "/Library/PrivilegedHelperTools/com.imanmontajabi.rat.helper"
    static func plistPath(_ label: String) -> String { "/Library/LaunchDaemons/\(label).plist" }

    let listURL: URL
    let snapshotURL: URL
    private var lastWritten: Set<String>?

    /// `helperInstalled` and `autoApply` can be forced so tests do not depend on (or touch) the machine's real state.
    init(supportDirectory: URL, helperInstalled: Bool? = nil, autoApply: Bool? = nil) {
        listURL = supportDirectory.appendingPathComponent("blocklist.txt")
        snapshotURL = supportDirectory.appendingPathComponent("system-sockets.txt")
        self.helperInstalled = helperInstalled ?? FileManager.default.fileExists(atPath: Self.plistPath(Self.applyLabel))
        self.autoApply = autoApply ?? UserDefaults.standard.bool(forKey: "autoApplyPF")
    }

    func needsApply(for wanted: Set<String>) -> Bool { wanted != applied }

    /// Called whenever the wanted blocklist may have changed. With the helper installed and auto-apply on,
    /// writing the file is the whole job: the daemon watches it.
    func sync(_ wanted: Set<String>) {
        guard helperInstalled, autoApply, wanted != lastWritten else { return }
        if writeList(wanted) { applied = wanted }
    }

    /// One-off apply behind an admin prompt (used when the helper is not installed).
    func applyOnce(_ wanted: Set<String>) {
        guard !busy else { return }
        guard let helper = Bundle.main.url(forResource: "rat-pf-helper", withExtension: "sh")?.path else {
            lastError = "Helper script missing from the app bundle."
            return
        }
        guard writeList(wanted) else { return }
        let command = "/bin/sh \(Self.shellQuote(helper)) apply \(Self.shellQuote(listURL.path))"
        runPrivileged(command) { ok in if ok { self.applied = wanted } }
    }

    func installHelper() {
        guard let src = Bundle.main.url(forResource: "rat-pf-helper", withExtension: "sh")?.path else {
            lastError = "Helper script missing from the app bundle."
            return
        }
        let owner = NSUserName()
        func plist(_ label: String, _ args: [String], extra: String) -> String {
            """
            <?xml version="1.0" encoding="UTF-8"?>
            <!DOCTYPE plist PUBLIC "-//Apple//DTD PLIST 1.0//EN" "http://www.apple.com/DTDs/PropertyList-1.0.dtd">
            <plist version="1.0"><dict>
            <key>Label</key><string>\(label)</string>
            <key>ProgramArguments</key><array>\(args.map { "<string>\(xml($0))</string>" }.joined())</array>
            \(extra)
            </dict></plist>
            """
        }
        let apply = plist(Self.applyLabel, [Self.helperPath, "apply", listURL.path],
                          extra: "<key>WatchPaths</key><array><string>\(xml(listURL.path))</string></array><key>RunAtLoad</key><true/>")
        let sampler = plist(Self.samplerLabel, [Self.helperPath, "snapshot", snapshotURL.path, owner],
                            extra: "<key>StartInterval</key><integer>5</integer><key>RunAtLoad</key><true/>")
        // The owner of the files the daemon writes into the user's support directory is the user, never root.
        let script = """
        set -e
        mkdir -p /Library/PrivilegedHelperTools
        install -m 755 -o root -g wheel \(Self.shellQuote(src)) \(Self.shellQuote(Self.helperPath))
        cat > \(Self.shellQuote(Self.plistPath(Self.applyLabel))) <<'PLIST'
        \(apply)
        PLIST
        cat > \(Self.shellQuote(Self.plistPath(Self.samplerLabel))) <<'PLIST'
        \(sampler)
        PLIST
        for f in \(Self.shellQuote(Self.plistPath(Self.applyLabel))) \(Self.shellQuote(Self.plistPath(Self.samplerLabel))); do
          chown root:wheel "$f"; chmod 644 "$f"
        done
        for l in \(Self.applyLabel) \(Self.samplerLabel); do /bin/launchctl bootout system/$l 2>/dev/null || true; done
        /bin/launchctl bootstrap system \(Self.shellQuote(Self.plistPath(Self.applyLabel)))
        /bin/launchctl bootstrap system \(Self.shellQuote(Self.plistPath(Self.samplerLabel)))
        """
        runPrivilegedScript(script) { ok in
            if ok { self.helperInstalled = true; self.autoApply = true; self.lastWritten = nil }
        }
    }

    func removeHelper() {
        let script = """
        for l in \(Self.applyLabel) \(Self.samplerLabel); do /bin/launchctl bootout system/$l 2>/dev/null || true; done
        /sbin/pfctl -a com.apple/rat -F all >/dev/null 2>&1 || true
        rm -f \(Self.shellQuote(Self.plistPath(Self.applyLabel))) \(Self.shellQuote(Self.plistPath(Self.samplerLabel))) \(Self.shellQuote(Self.helperPath))
        """
        runPrivilegedScript(script) { ok in
            if ok { self.helperInstalled = false; self.autoApply = false; self.applied = []; self.lastWritten = nil }
        }
    }

    /// Latest root snapshot of all processes' sockets, if the helper is installed and the file is fresh.
    func systemSnapshotText() -> String? {
        guard helperInstalled,
              let attrs = try? FileManager.default.attributesOfItem(atPath: snapshotURL.path),
              let modified = attrs[.modificationDate] as? Date, Date().timeIntervalSince(modified) < 30 else { return nil }
        return try? String(contentsOf: snapshotURL, encoding: .utf8)
    }

    // MARK: plumbing

    @discardableResult
    private func writeList(_ ips: Set<String>) -> Bool {
        do {
            try ips.sorted().joined(separator: "\n").appending("\n").write(to: listURL, atomically: true, encoding: .utf8)
            lastWritten = ips
            logger.notice("blocklist written: \(ips.count, privacy: .public) addresses")
            return true
        } catch {
            lastError = error.localizedDescription
            logger.error("could not write blocklist: \(error.localizedDescription, privacy: .public)")
            return false
        }
    }

    private func runPrivilegedScript(_ script: String, done: @escaping (Bool) -> Void) {
        let url = FileManager.default.temporaryDirectory.appendingPathComponent("rat-\(UUID().uuidString).sh")
        do { try script.write(to: url, atomically: true, encoding: .utf8) } catch { lastError = error.localizedDescription; return }
        runPrivileged("/bin/sh \(Self.shellQuote(url.path))") { ok in
            try? FileManager.default.removeItem(at: url)
            done(ok)
        }
    }

    private func runPrivileged(_ command: String, done: @escaping (Bool) -> Void) {
        guard !busy else { return }
        busy = true
        lastError = nil
        Task.detached {
            let p = Process()
            p.executableURL = URL(fileURLWithPath: "/usr/bin/osascript")
            p.arguments = ["-e", "do shell script \"\(Self.appleScriptEscape(command))\" with administrator privileges with prompt \"Rat needs permission to update the firewall.\""]
            let err = Pipe()
            p.standardError = err
            p.standardOutput = FileHandle.nullDevice
            try? p.run()
            p.waitUntilExit()
            let msg = String(decoding: err.fileHandleForReading.readDataToEndOfFile(), as: UTF8.self)
            await MainActor.run {
                self.busy = false
                if p.terminationStatus == 0 {
                    logger.notice("privileged step succeeded")
                    done(true)
                } else {
                    self.lastError = msg.contains("-128") ? "Cancelled." : msg.trimmingCharacters(in: .whitespacesAndNewlines)
                    logger.error("privileged step failed")
                    done(false)
                }
            }
        }
    }

    nonisolated private func xml(_ s: String) -> String {
        s.replacingOccurrences(of: "&", with: "&amp;").replacingOccurrences(of: "<", with: "&lt;").replacingOccurrences(of: ">", with: "&gt;")
    }

    nonisolated static func shellQuote(_ s: String) -> String { "'" + s.replacingOccurrences(of: "'", with: "'\\''") + "'" }

    nonisolated static func appleScriptEscape(_ s: String) -> String {
        s.replacingOccurrences(of: "\\", with: "\\\\").replacingOccurrences(of: "\"", with: "\\\"")
    }
}

enum TimeRange: String, CaseIterable, Identifiable {
    case fiveMinutes = "5m", hour = "1h", day = "1d", week = "1w"
    var id: String { rawValue }
    var seconds: TimeInterval {
        switch self {
        case .fiveMinutes: return 300
        case .hour: return 3600
        case .day: return 86_400
        case .week: return 604_800
        }
    }
    var title: String {
        switch self {
        case .fiveMinutes: return "5 minutes"
        case .hour: return "1 hour"
        case .day: return "1 day"
        case .week: return "1 week"
        }
    }
}

struct AppEntry: Identifiable {

final class Ledger: ObservableObject {

    @Published var history: [TrafficPoint] = []
    @Published var sessionIn: UInt64 = 0
    @Published var sessionOut: UInt64 = 0
    @Published var rateIn = 0.0
    @Published var rateOut = 0.0
    @Published var myLocation: GeoPoint?
    @Published var deniedCount = 0
    @Published var unconfirmedCount = 0
    @Published var incomingCount = 0
    @Published var systemWide = false

    let rules: RulesModel
    let enforcer: Enforcer
    let geo = GeoService()
    private(set) lazy var alerts: AlertCenter = {
        let center = AlertCenter(rules: rules) { [unowned self] alert, verdict in
            if verdict == .deny { recordDenied(appID: alert.appID, address: alert.address, host: alert.host, port: alert.port, proto: alert.proto) }
            refreshBlocklist()
        }
        center.isStillPending = { [unowned self] a in
            let c = Connection(process: ProcessIdentity(path: a.appID), host: a.host == a.address ? nil : a.host, address: a.address, port: a.port,
                               proto: a.proto == "UDP" ? .udp : .tcp)
            return rules.engine.evaluate(c) == .ask
        }
        return center
    }()
    var onTick: (() -> Void)?

    private let names = HostNames()
    private let store: StateStore
    private var previous: [Int32: TrafficCounter] = [:]
    private var table: [String: AppEntry] = [:]
    private var buckets: [Date: Bucket] = [:]
    private var denied: [String: [String: DeniedEntry]] = [:]
    private var signingCache: [String: (identifier: String?, team: String?)] = [:]
    private var running = false
    private var demo = false
    private var lastSave = Date()
    private var lastTick = Date()
    private var startedAt = Date()

    init(directory: URL? = nil) {
        #if DEBUG
        // Demo mode never reads or writes the real data folder.
        let demoDir = ProcessInfo.processInfo.environment["RAT_DEMO_MAP"] != nil ? FileManager.default.temporaryDirectory.appendingPathComponent("rat-demo-\(UUID())") : nil
        store = StateStore.standard(base: directory ?? demoDir)
        #else
        store = StateStore.standard(base: directory)
        #endif
        rules = RulesModel(directory: store.directory)
        enforcer = Enforcer(supportDirectory: store.directory)
        myLocation = GeoService.localPoint()
        let saved = store.load()
        for a in saved.apps where !a.id.hasPrefix("pid ") {
            // Re-resolve plain executables so names learned by older versions ("2.1.308") are corrected and merged.
            let ident = a.bundlePath != nil || a.isSystem ? AppIdentity(id: a.id, name: a.name, bundlePath: a.bundlePath, isSystem: a.isSystem, isExtension: a.id.hasSuffix("#ext"))
                                                         : AppIdentity.resolve(processPath: a.id)
            var e = table[ident.id] ?? AppEntry(identity: ident)
            e.totalIn += a.totalIn; e.totalOut += a.totalOut; e.lastActive = max(e.lastActive, a.lastActive)
            table[ident.id] = e
        }
        for (app, entries) in saved.denied { denied[app] = Dictionary(uniqueKeysWithValues: entries.map { ($0.address, $0) }) }
        for b in saved.buckets { buckets[b.minute] = b }
        apps = Array(table.values)
        alerts.iconProvider = { [weak self] id in self?.table[id]?.identity.icon ?? NSImage() }
        #if DEBUG
        if ProcessInfo.processInfo.environment["RAT_DEMO_MAP"] != nil { installDemoData() }
        #endif
    }

    // MARK: lifecycle

    func start() {
        guard !running else { return }
        running = true
        startedAt = Date()
        logger.notice("ledger started")
        Task { await loop() }
    }

    func save() {
        guard !demo else { return }   // demo data must never reach the real state files
        let weekAgo = Date().addingTimeInterval(-TimeRange.week.seconds)
        let state = PersistedState(
            apps: table.values.map { PersistedApp(id: $0.id, name: $0.identity.name, bundlePath: $0.identity.bundlePath, isSystem: $0.identity.isSystem, totalIn: $0.totalIn, totalOut: $0.totalOut, lastActive: $0.lastActive) },
            denied: denied.mapValues { Array($0.values) },
            buckets: buckets.values.filter { $0.minute > weekAgo }.sorted { $0.minute < $1.minute })
        store.save(state)
        rules.saveNow()
        lastSave = Date()
    }

    /// Re-checks queued alerts against the current rules and mode: answered, allowed or denied apps never prompt.
    func revalidateAlerts() { alerts.revalidate() }

    func shutdown() {
        rules.purgeSessionRules()
        save()
    }

    private func loop() async {
        while running && !demo {
            let useSnapshot = enforcer.systemSnapshotText()
            let (conns, traffic) = await Task.detached(priority: .utility) { () -> ([ObservedConnection], [TrafficCounter]) in
                var c = ConnectionSampler.sample()
                if let text = useSnapshot {
                    // Root helper snapshot: adds processes this app cannot inspect (other users, system daemons).
                    var seen = Set(c.map { "\($0.pid)|\($0.proto.rawValue)|\($0.localPort)|\($0.remoteAddress)|\($0.remotePort)" })
                    for x in SystemSnapshot.parse(text) where seen.insert("\(x.pid)|\(x.proto.rawValue)|\(x.localPort)|\(x.remoteAddress)|\(x.remotePort)").inserted { c.append(x) }
                }
                return (c, TrafficSampler.sample())
            }.value
            systemWide = useSnapshot != nil
            await ingest(conns, traffic)
            rules.refreshNetwork()
            refreshBlocklist()
            alerts.revalidate()
            onTick?()
            if Date().timeIntervalSince(lastSave) > 30 { save() }
            try? await Task.sleep(for: .seconds(1.5))
        }
    }

    // MARK: sampling

    private func signing(for identity: AppIdentity) -> (identifier: String?, team: String?) {
        if let c = signingCache[identity.id] { return c }
        let s = identity.signing()
        signingCache[identity.id] = s
        return s
    }

    private func ingest(_ conns: [ObservedConnection], _ traffic: [TrafficCounter]) async {
        let now = Date()
        let dt = max(now.timeIntervalSince(lastTick), 0.2)
        lastTick = now
        let warmedUp = now.timeIntervalSince(startedAt) > 12   // don't alert for sockets that already existed at launch

        var identityByPid: [Int32: AppIdentity] = [:]
        var perApp: [String: (conns: [ObservedConnection], listening: Int)] = [:]
        for c in conns {
            guard !c.processPath.hasPrefix("pid ") else { continue }   // process already gone: nothing to attribute
            let ident = identityByPid[c.pid] ?? AppIdentity.resolve(processPath: c.processPath, pid: c.pid)
            identityByPid[c.pid] = ident
            var e = perApp[ident.id] ?? ([], 0)
            if c.tcpState == "LISTEN" { e.listening += 1 } else if c.isOutbound { e.conns.append(c) }
            perApp[ident.id] = e
            if table[ident.id] == nil { table[ident.id] = AppEntry(identity: ident) }
            table[ident.id]!.pids.insert(c.pid)
        }

        var rxByApp: [String: Double] = [:], txByApp: [String: Double] = [:]
        var totalRx = 0.0, totalTx = 0.0
        for t in traffic {
            defer { previous[t.pid] = t }
            guard let p = previous[t.pid], t.bytesIn >= p.bytesIn, t.bytesOut >= p.bytesOut else { continue }
            let dIn = Double(t.bytesIn - p.bytesIn), dOut = Double(t.bytesOut - p.bytesOut)
            totalRx += dIn; totalTx += dOut
            let ident: AppIdentity
            if let i = identityByPid[t.pid] { ident = i } else {
                let path = ConnectionSampler.path(of: t.pid)
                guard !path.hasPrefix("pid ") else { continue }
                ident = AppIdentity.resolve(processPath: path, pid: t.pid)
                if table[ident.id] == nil { table[ident.id] = AppEntry(identity: ident) }
            }
            rxByApp[ident.id, default: 0] += dIn; txByApp[ident.id, default: 0] += dOut
            table[ident.id]!.totalIn += UInt64(dIn); table[ident.id]!.totalOut += UInt64(dOut)
            table[ident.id]!.pids.insert(t.pid)
        }
        sessionIn += UInt64(totalRx); sessionOut += UInt64(totalTx)
        rateIn = totalRx / dt; rateOut = totalTx / dt
        history.append(TrafficPoint(date: now, rx: totalRx, tx: totalTx))
        if history.count > 2400 { history.removeFirst(history.count - 2400) }
        let minute = Date(timeIntervalSince1970: (now.timeIntervalSince1970 / 60).rounded(.down) * 60)
        var b = buckets[minute] ?? Bucket(minute: minute, rx: 0, tx: 0)
        b.rx += totalRx; b.tx += totalTx
        buckets[minute] = b

        var denyCount = 0, askCount = 0, listenCount = 0
        for id in table.keys {
            var e = table[id]!
            let rx = rxByApp[id] ?? 0, tx = txByApp[id] ?? 0
            e.rates.append((rx, tx)); if e.rates.count > 14 { e.rates.removeFirst() }
            if rx + tx > 0 || perApp[id] != nil { e.lastActive = now }
            e.listening = perApp[id]?.listening ?? 0
            listenCount += e.listening
            if perApp[id] == nil { e.pids = e.pids.filter { pid in traffic.contains { $0.pid == pid } } }
            e.otherUsers = !e.pids.isEmpty && e.pids.allSatisfy { ownerUID(of: $0) != getuid() }

            var remotes: [RemoteHost] = []
            var seen = Set<String>()
            let sig = e.remotes.isEmpty && perApp[id] == nil ? (nil as String?, nil as String?) : signing(for: e.identity)
            for c in perApp[id]?.conns ?? [] where seen.insert("\(c.remoteAddress):\(c.remotePort)").inserted {
                let host = await names.name(for: c.remoteAddress)
                let geoPoint = isLocalAddress(c.remoteAddress) ? nil : geo.lookup(c.remoteAddress)
                remotes.append(RemoteHost(address: c.remoteAddress, host: host, port: c.remotePort, proto: c.proto.rawValue.uppercased(), geo: geoPoint))

                guard !isLocalAddress(c.remoteAddress) else { continue }
                                let conn = Connection(process: ProcessIdentity(signingID: sig.0, teamID: sig.1, path: e.identity.ruleKey), host: host == c.remoteAddress ? nil : host,
                                      address: c.remoteAddress, port: c.remotePort, proto: c.proto == .udp ? .udp : .tcp)
                let matched = rules.engine.matchingRule(conn)
                let verdict = matched?.verdict ?? rules.engine.defaultVerdict
                if matched == nil, verdict != .ask, warmedUp { noteSilentConnection(app: e.identity.name, host: host, allowed: verdict == .allow) }
                switch verdict {
                case .deny:
                    denyCount += 1
                    recordDenied(appID: id, address: c.remoteAddress, host: host, port: c.remotePort, proto: remotes.last!.proto)
                case .ask:
                    askCount += 1
                    if warmedUp {
                        alerts.report(ConnectionAlert(appID: e.identity.ruleKey, appName: e.identity.name, host: host, address: c.remoteAddress, port: c.remotePort, proto: remotes.last!.proto))
                    }
                case .allow: break
                }
            }
            e.remotes = remotes
            table[id] = e
        }
        deniedCount = denyCount; unconfirmedCount = askCount; incomingCount = listenCount
        apps = Array(table.values)
    }

    private var uidCache: [Int32: uid_t] = [:]

    private func ownerUID(of pid: Int32) -> uid_t {
        if let u = uidCache[pid] { return u }
        var info = proc_bsdinfo()
        let n = proc_pidinfo(pid, PROC_PIDTBSDINFO, 0, &info, Int32(MemoryLayout<proc_bsdinfo>.size))
        let uid = n == Int32(MemoryLayout<proc_bsdinfo>.size) ? info.pbi_uid : getuid()
        uidCache[pid] = uid
        return uid
    }

    private var notifiedSilent = Set<String>()

    /// "Silent Mode Connections": tell the user the first time a connection was allowed or denied without asking.
    private func noteSilentConnection(app: String, host: String, allowed: Bool) {
        guard AppSettings.shared.notifySilentConnections, notifiedSilent.insert("\(app)|\(registrableDomain(host))").inserted else { return }
        Notifier.post(title: allowed ? "Silent Mode: connection allowed" : "Silent Mode: connection denied", body: "\(app) → \(host)")
    }

    // MARK: blocking

    /// Allow/Deny from the process list. Deny also remembers the server addresses the app is using right now.
    func setPolicy(_ p: Policy?, for appID: String) {
        rules.setPolicy(p, for: table[appID]?.identity.ruleKey ?? appID)
        alerts.revalidate()
        if p == .deny, let e = table[appID] {
            for r in e.remotes { recordDenied(appID: appID, address: r.address, host: r.host, port: r.port, proto: r.proto) }
        }
        refreshBlocklist()
    }

    private func recordDenied(appID: String, address: String, host: String, port: Int, proto: String) {
        guard !isLocalAddress(address) else { return }
        var set = denied[appID] ?? [:]
        guard set.count < 2000 else { return }
        set[address] = DeniedEntry(address: address, host: host, port: port, proto: proto)
        denied[appID] = set
    }

    /// Every address that should be blocked right now: remembered denied connections that the rules still deny.
    var wantedBlocklist: Set<String> {
        var out = Set<String>()
        for (appID, entries) in denied {
            let sig = table[appID].map { signing(for: $0.identity) } ?? (nil, nil)
            let key = table[appID]?.identity.ruleKey ?? appID
            for e in entries.values {
                let c = Connection(process: ProcessIdentity(signingID: sig.0, teamID: sig.1, path: key), host: e.host == e.address ? nil : e.host,
                                   address: e.address, port: e.port, proto: e.proto == "UDP" ? .udp : .tcp)
                if rules.engine.evaluate(c) == .deny { out.insert(e.address) }
            }
        }
        return out
    }

    /// With the root helper installed, keep the pf table in step with the rules automatically.
    func refreshBlocklist() { enforcer.sync(wantedBlocklist) }

    func applyBlocking() { enforcer.applyOnce(wantedBlocklist) }

    func blockedAddresses(for appID: String) -> Int {
        let wanted = wantedBlocklist
        return denied[appID]?.keys.filter(wanted.contains).count ?? 0
    }

    // MARK: derived data

    func domains() -> [(name: String, count: Int)] {
        var d: [String: Int] = [:]
        for a in apps { for r in a.remotes { d[registrableDomain(r.host), default: 0] += 1 } }
        return d.map { ($0.key, $0.value) }.sorted { $0.1 > $1.1 }
    }

    func countries() -> [(name: String, count: Int)] {
        var d: [String: Int] = [:]
        for a in apps { for r in a.remotes { if let c = r.geo?.country { d[c, default: 0] += 1 } } }
        return d.map { ($0.key, $0.value) }.sorted { $0.1 > $1.1 }
    }

    /// Bars for the traffic chart. The window shrinks to the data that exists (never a lone sliver on the right edge)
    /// and the bars are sized so that they tile it completely.
    func chartLayout(_ range: TimeRange, now: Date = Date()) -> ChartLayout {
        let windowStart = now.addingTimeInterval(-range.seconds)
        // (time, rx, tx) samples: live samples up to an hour, minute buckets beyond.
        let samples: [(Date, Double, Double)] = range.seconds <= 3600
            ? history.filter { $0.date >= windowStart }.map { ($0.date, $0.rx, $0.tx) }
            : buckets.values.filter { $0.minute >= windowStart }.map { ($0.minute, $0.rx, $0.tx) }
        let minSpan = min(range.seconds, 60.0)
        let first = samples.map(\.0).min() ?? now.addingTimeInterval(-minSpan)
        let start = min(max(windowStart, first), now.addingTimeInterval(-minSpan))
        let span = now.timeIntervalSince(start)
        let floorSize: TimeInterval = range.seconds <= 3600 ? 1.5 : 60
        let size = max(span / 60, floorSize)
        let count = max(Int((span / size).rounded(.up)), 1)
        var rx = [Double](repeating: 0, count: count), tx = [Double](repeating: 0, count: count)
        for (t, r, x) in samples {
            let i = min(max(Int(t.timeIntervalSince(start) / size), 0), count - 1)
            rx[i] += r; tx[i] += x
        }
        let starts = (0..<count).map { start.addingTimeInterval(Double($0) * size) }
        let ends = (0..<count).map { start.addingTimeInterval(Double($0 + 1) * size) }
        // Square-root scale so quiet traffic stays visible next to a burst.
        for i in 0..<count { rx[i] = rx[i].squareRoot(); tx[i] = tx[i].squareRoot() }
        let bars = zip(zip(starts, ends), zip(rx, tx)).map { ChartBar(start: $0.0.0, end: $0.0.1, rx: $0.1.0, tx: $0.1.1) }
        let peak = max((rx + tx).max() ?? 0, 1) * 1.1
        let format: Date.FormatStyle = range == .week ? .dateTime.weekday(.abbreviated).day()
            : range == .fiveMinutes ? .dateTime.hour().minute().second() : .dateTime.hour().minute()
        return ChartLayout(bars: bars, start: start, end: start.addingTimeInterval(Double(count) * size), peak: peak, labelFormat: format)
    }

func formatBytes(_ n: UInt64) -> String {
    let f = ByteCountFormatter()

    f.allowsNonnumericFormatting = false   // "0 bytes" instead of "Zero KB"

struct PersistedApp: Codable {
    var id: String
    var name: String
    var bundlePath: String?
    var isSystem: Bool
    var totalIn: UInt64
    var totalOut: UInt64
    var lastActive: Date
}

struct Bucket: Codable {
    var minute: Date
    var rx: Double
    var tx: Double
}

/// A connection the user denied; re-evaluated against the rules so the blocklist shrinks when a rule is removed.
struct DeniedEntry: Codable {
    var address: String
    var host: String
    var port: Int
    var proto: String
}

struct PersistedState: Codable {
    var apps: [PersistedApp] = []
    var denied: [String: [DeniedEntry]] = [:]
    var buckets: [Bucket] = []

    init(apps: [PersistedApp] = [], denied: [String: [DeniedEntry]] = [:], buckets: [Bucket] = []) {
        self.apps = apps; self.denied = denied; self.buckets = buckets
    }

    // Tolerant of state files written by earlier versions.
    init(from d: Decoder) throws {
        let c = try d.container(keyedBy: CodingKeys.self)
        apps = try c.decodeIfPresent([PersistedApp].self, forKey: .apps) ?? []
        denied = try c.decodeIfPresent([String: [DeniedEntry]].self, forKey: .denied) ?? [:]
        buckets = try c.decodeIfPresent([Bucket].self, forKey: .buckets) ?? []
    }
}

/// JSON state in ~/Library/Application Support/com.imanmontajabi.Rat (the only folder Rat writes to).
struct StateStore {
    let directory: URL
    var url: URL { directory.appendingPathComponent("state.json") }

    static func standard(base: URL? = nil) -> StateStore {
        let root = base ?? FileManager.default.urls(for: .applicationSupportDirectory, in: .userDomainMask)[0].appendingPathComponent("com.imanmontajabi.Rat")
        try? FileManager.default.createDirectory(at: root, withIntermediateDirectories: true)
        return StateStore(directory: root)
    }

    func load() -> PersistedState {
        guard let data = try? Data(contentsOf: url) else { return PersistedState() }
        let dec = JSONDecoder()
        dec.dateDecodingStrategy = .iso8601
        do { return try dec.decode(PersistedState.self, from: data) } catch {
            logger.error("state unreadable, starting fresh: \(error.localizedDescription, privacy: .public)")
            return PersistedState()
        }
    }

    func save(_ state: PersistedState) {
        do {
            let enc = JSONEncoder()
            enc.dateEncodingStrategy = .iso8601
            try enc.encode(state).write(to: url, options: .atomic)
        } catch {
            logger.error("state save failed: \(error.localizedDescription, privacy: .public)")
        }
    }
}

/// The rule engine, its persistence and everything the Rules window edits.
@MainActor
final class RulesModel: ObservableObject {
    @Published var ruleSet: RuleSet { didSet { engine.ruleSet = ruleSet; scheduleSave() } }
    @Published private(set) var currentNetwork: NetworkWatcher.Network?
    @Published var lastMessage: String?

    let engine: RuleEngine
    private let store: RuleStore
    private var saveTask: Task<Void, Never>?

    init(directory: URL) {
        store = RuleStore(url: directory.appendingPathComponent("rules.json"))
        var set = (try? store.loadSet()) ?? RuleSet.standard()
        if !set.profiles.contains(where: { $0.id == set.activeProfileID }), let first = set.profiles.first { set.activeProfileID = first.id }
        Self.repair(&set)
        ruleSet = set
        engine = RuleEngine(ruleSet: set)
    }

    /// Fixes rules written by earlier versions: an IP address saved as a "domain" (it could never match), and
    /// versioned executables ("…/claude/versions/2.1.308") that should refer to the tool, not one version.
    static func repair(_ set: inout RuleSet) {
        for i in set.rules.indices {
            if let d = set.rules[i].remoteDomain, isIPLiteral(d) {
                set.rules[i].remoteAddress = d
                set.rules[i].remoteDomain = nil
            }
            if let p = set.rules[i].processPath, !p.contains(".app") {
                let resolved = AppIdentity.resolve(processPath: p).id
                if resolved != p && resolved != "macOS" { set.rules[i].processPath = resolved }
            }
        }
    }

    // MARK: profiles

    var activeProfile: Profile? { ruleSet.profiles.first { $0.id == ruleSet.activeProfileID } }

    func activate(_ id: UUID) {
        guard ruleSet.activeProfileID != id else { return }
        ruleSet.activeProfileID = id
        logger.notice("profile switched")
    }

    func addProfile(named name: String) {
        let n = name.trimmingCharacters(in: .whitespaces)
        guard !n.isEmpty else { return }
        ruleSet.profiles.append(Profile(name: n))
    }

    func removeProfile(_ id: UUID) {
        guard ruleSet.profiles.count > 1 else { return }
        ruleSet.profiles.removeAll { $0.id == id }
        ruleSet.triggers.removeAll { $0.profileID == id }
        ruleSet.rules.removeAll { $0.profileID == id }
        if ruleSet.activeProfileID == id, let first = ruleSet.profiles.first { ruleSet.activeProfileID = first.id }
    }

    /// Polls the network; with automatic switching on, a known network selects its profile.
    func refreshNetwork() { applyNetwork(NetworkWatcher.current()) }

    func applyNetwork(_ net: NetworkWatcher.Network?) {
        guard net != currentNetwork else { return }
        currentNetwork = net
        guard ruleSet.automaticProfileSwitching, let net else { return }
        if let target = engine.profile(forNetwork: net.signature) {
            switchAutomatically(to: target, network: net)
            return
        }
        switch ruleSet.unknownNetworkAction {
        case .ask: askForProfile(network: net)
        case .activate(let id): if ruleSet.profiles.contains(where: { $0.id == id }) { switchAutomatically(to: id, network: net) }
        case .doNothing, .deactivate: break
        }
    }

    private func switchAutomatically(to id: UUID, network: NetworkWatcher.Network) {
        guard ruleSet.activeProfileID != id else { return }
        activate(id)
        if AppSettings.shared.notifyProfileSwitch, let name = activeProfile?.name {
            Notifier.post(title: "Profile switched to “\(name)”", body: "Joined \(network.name)")
        }
    }

    /// "When joining an unknown network: Ask".
    private func askForProfile(network: NetworkWatcher.Network) {
        let alert = NSAlert()
        alert.messageText = "Joined “\(network.name)”"
        alert.informativeText = "Which profile should be active on this network? Rat will switch to it automatically next time."
        let popup = NSPopUpButton(frame: NSRect(x: 0, y: 0, width: 240, height: 26))
        for p in ruleSet.profiles { popup.addItem(withTitle: p.name) }
        popup.selectItem(withTitle: activeProfile?.name ?? "")
        alert.accessoryView = popup
        alert.addButton(withTitle: "Use Profile")
        alert.addButton(withTitle: "Not Now")
        NSApp.activate(ignoringOtherApps: true)
        guard alert.runModal() == .alertFirstButtonReturn, popup.indexOfSelectedItem >= 0, popup.indexOfSelectedItem < ruleSet.profiles.count else { return }
        let target = ruleSet.profiles[popup.indexOfSelectedItem].id
        rememberCurrentNetwork(profile: target)
        activate(target)
    }

    func rememberCurrentNetwork(profile: UUID) {
        guard let net = currentNetwork else { return }
        ruleSet.triggers.removeAll { $0.signature == net.signature }
        ruleSet.triggers.append(NetworkTrigger(signature: net.signature, name: net.name, profileID: profile))
    }

    // MARK: silent mode

    var silentMode: SilentMode {
        get { switch ruleSet.silentVerdict { case .allow?: return .allow; case .deny?: return .deny; default: return .off } }
        set { ruleSet.silentVerdict = newValue == .allow ? .allow : newValue == .deny ? .deny : nil }
    }

    // MARK: per-app policy (the allow/deny buttons in the process list)

    private func isWholeAppRule(_ r: Rule, _ appID: String) -> Bool {
        r.processPath == appID && r.remoteDomain == nil && r.remoteHost == nil && r.remoteAddress == nil && r.ports == nil
            && r.proto == nil && r.signingID == nil && r.teamID == nil
    }

    func policy(for appID: String) -> Policy? {
        let rule = ruleSet.rules.first {
            isWholeAppRule($0, appID) && $0.enabled && ($0.profileID == nil || $0.profileID == ruleSet.activeProfileID)
        }
        return rule.map { $0.verdict == .deny ? .deny : .allow }
    }

    func setPolicy(_ p: Policy?, for appID: String) {
        ruleSet.rules.removeAll { isWholeAppRule($0, appID) && ($0.profileID == nil || $0.profileID == ruleSet.activeProfileID) }
        guard let p else { return }
        ruleSet.rules.append(Rule(verdict: p == .allow ? .allow : .deny, processPath: appID, approved: !ruleSet.markNewRulesAsUnapproved,
                                  profileID: ruleSet.activeProfileID, note: "From the process list"))
    }

    func rules(for appID: String) -> [Rule] {
        ruleSet.rules.filter { $0.processPath == appID }
    }

    // MARK: alert answers

    enum Scope: String, CaseIterable, Identifiable {
        case any = "This App (Any Server)", domain = "Domain", host = "Exact host", port = "Port"
        var id: String { rawValue }
    }

    func addRule(from alert: ConnectionAlert, verdict: Verdict, scope: Scope, lifetime: Lifetime) {
        // "Once" allow needs no rule. "Once" deny is a session rule so the open connection is really cut.
        if lifetime == .once && verdict == .allow { return }
        var r = Rule(verdict: verdict, processPath: alert.appID, approved: !ruleSet.markNewRulesAsUnapproved,
                     lifetime: lifetime == .once ? .untilQuit : lifetime, profileID: ruleSet.activeProfileID, note: "From a connection alert")
        switch scope {
        case .any: break
        case .domain:
            // A raw IP has no domain: scope it to the address instead.
            if !isIPLiteral(alert.host) { r.remoteDomain = registrableDomain(alert.host) } else { r.remoteAddress = alert.address }
        case .host: if !isIPLiteral(alert.host) { r.remoteHost = alert.host } else { r.remoteAddress = alert.address }
        case .port: r.ports = alert.port...alert.port; r.proto = alert.proto == "UDP" ? .udp : .tcp
        }
        if lifetime == .once { r.remoteAddress = alert.address; r.ports = alert.port...alert.port }
        ruleSet.rules.append(r)
    }

    // MARK: rule manager

    var unapprovedCount: Int { ruleSet.rules.filter { !$0.approved }.count }

    func approve(_ id: UUID) {
        guard let i = ruleSet.rules.firstIndex(where: { $0.id == id }) else { return }
        ruleSet.rules[i].approved = true; ruleSet.rules[i].enabled = true
    }

    func approveAll() {
        for i in ruleSet.rules.indices where !ruleSet.rules[i].approved { ruleSet.rules[i].approved = true; ruleSet.rules[i].enabled = true }
    }

    func reject(_ id: UUID) { ruleSet.rules.removeAll { $0.id == id } }
    func rejectAllUnapproved() { ruleSet.rules.removeAll { !$0.approved } }

    func upsert(_ rule: Rule) {
        if let i = ruleSet.rules.firstIndex(where: { $0.id == rule.id }) { ruleSet.rules[i] = rule } else { ruleSet.rules.append(rule) }
    }

    func exportRules(to url: URL) {
        do {
            try engine.exportRules().encoded().write(to: url, options: .atomic)
            lastMessage = "Exported \(engine.exportRules().rules.count) rules."
        } catch { lastMessage = "Export failed: \(error.localizedDescription)" }
    }

    func importRules(from url: URL) {
        do {
            let export = try RuleExport.decode(Data(contentsOf: url))
            let before = ruleSet.rules.count
            engine.ruleSet = ruleSet
            let added = engine.importRules(export, markUnapproved: true)
            ruleSet = engine.ruleSet
            lastMessage = "Imported \(added) rules (\(export.rules.count - added) already present). Review and approve them to activate."
            _ = before
        } catch { lastMessage = "Import failed: \(error.localizedDescription)" }
    }

    // MARK: persistence

    func purgeSessionRules() {
        ruleSet.rules.removeAll { $0.lifetime != .forever }
    }

    func saveNow() {
        saveTask?.cancel()
        do { try store.saveSet(ruleSet) } catch { logger.error("rules save failed: \(error.localizedDescription, privacy: .public)") }
    }

    private func scheduleSave() {
        saveTask?.cancel()
        saveTask = Task { [weak self] in
            try? await Task.sleep(for: .milliseconds(400))
            if !Task.isCancelled { self?.saveNow() }
        }
    }
}

private let logger = Logger(subsystem: "com.imanmontajabi.Rat", category: "update")

/// Asks the GitHub Releases API whether a newer version exists. Runs only when the user chooses it.
enum UpdateChecker {
    static let repo = "ImanMontajabi/Rat"

    /// True when `latest` is a higher dotted version than `current` ("v0.3.0" vs "0.2.0").
    static func isNewer(_ latest: String, than current: String) -> Bool {
        func parts(_ s: String) -> [Int] { s.trimmingCharacters(in: CharacterSet(charactersIn: "vV")).split(separator: ".").map { Int($0) ?? 0 } }
        let a = parts(latest), b = parts(current)
        for i in 0..<max(a.count, b.count) {
            let x = i < a.count ? a[i] : 0, y = i < b.count ? b[i] : 0
            if x != y { return x > y }
        }
        return false
    }

    /// Silent check at launch when automatic checks are on and the last one is over a day old.
    @MainActor static func checkIfDue() async {
        let settings = AppSettings.shared
        guard settings.autoCheckUpdates else { return }
        if let last = settings.lastUpdateCheck, Date().timeIntervalSince(last) < 86_400 { return }
        if let release = await latest(includePreReleases: settings.showPreReleases),
           isNewer(release.tag, than: Bundle.main.infoDictionary?["CFBundleShortVersionString"] as? String ?? "0") {
            present(release, current: Bundle.main.infoDictionary?["CFBundleShortVersionString"] as? String ?? "0")
        }
    }

    struct Release { var tag: String; var url: String; var prerelease: Bool }

    /// The newest published release (optionally including pre-releases). Records the attempt in the settings.
    @MainActor static func latest(includePreReleases: Bool) async -> Release? {
        let settings = AppSettings.shared
        do {
            var req = URLRequest(url: URL(string: "https://api.github.com/repos/\(repo)/releases")!)
            req.setValue("application/vnd.github+json", forHTTPHeaderField: "Accept")
            let (data, _) = try await URLSession.shared.data(for: req)
            guard let list = try JSONSerialization.jsonObject(with: data) as? [[String: Any]] else { throw URLError(.cannotParseResponse) }
            settings.lastUpdateCheck = Date(); settings.lastUpdateCheckFailed = false
            return list.compactMap { j -> Release? in
                guard j["draft"] as? Bool != true, let tag = j["tag_name"] as? String else { return nil }
                let pre = j["prerelease"] as? Bool ?? false
                guard includePreReleases || !pre else { return nil }
                return Release(tag: tag, url: j["html_url"] as? String ?? "https://github.com/\(repo)/releases", prerelease: pre)
            }.max { !isNewer($1.tag, than: $0.tag) ? false : true }
        } catch {
            logger.error("update check failed: \(error.localizedDescription, privacy: .public)")
            settings.lastUpdateCheck = Date(); settings.lastUpdateCheckFailed = true
            return nil
        }
    }

    @MainActor private static func present(_ r: Release, current: String) {
        let alert = NSAlert()
        alert.messageText = "Rat \(r.tag) is available"
        alert.informativeText = "You have \(current)."
        alert.addButton(withTitle: "Open Download Page"); alert.addButton(withTitle: "Later")
        if alert.runModal() == .alertFirstButtonReturn, let u = URL(string: r.url) { NSWorkspace.shared.open(u) }
    }

    /// "Check Now": always answers.
    @MainActor static func check() async {
        let current = Bundle.main.infoDictionary?["CFBundleShortVersionString"] as? String ?? "0"
        let release = await latest(includePreReleases: AppSettings.shared.showPreReleases)
        if AppSettings.shared.lastUpdateCheckFailed {
            let alert = NSAlert(); alert.messageText = "Could not check for updates"; alert.informativeText = "Check your internet connection and try again."; alert.runModal(); return
        }
        guard let release else { let a = NSAlert(); a.messageText = "No releases found"; a.runModal(); return }
        if isNewer(release.tag, than: current) { present(release, current: current) }
        else { let a = NSAlert(); a.messageText = "Rat is up to date"; a.informativeText = "Version \(current) is the latest."; a.runModal() }
    }
}

/// Identifies the network the Mac is on, for automatic profile switching.
/// Uses the Wi-Fi name when macOS reveals it (it needs Location access) and otherwise the default gateway.
enum NetworkWatcher {
    struct Network: Equatable {
        var signature: String
        var name: String
    }

    static func current() -> Network? {
        if let ssid = CWWiFiClient.shared().interface()?.ssid(), !ssid.isEmpty { return Network(signature: ssid, name: ssid) }
        guard let store = SCDynamicStoreCreate(nil, "Rat" as CFString, nil, nil),
              let info = SCDynamicStoreCopyValue(store, "State:/Network/Global/IPv4" as CFString) as? [String: Any],
              let router = info["Router"] as? String else { return nil }
        return Network(signature: "gateway:\(router)", name: "Network via \(router)")
    }
}
